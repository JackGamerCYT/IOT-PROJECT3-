/* * ==================================================================================== * DỰ ÁN IOT: GIÁM SÁT NĂNG LƯỢNG & QUẢN LÝ TẢI THÔNG MINH (VERSION V13 - FULL SERIAL LOG) * VI ĐIỀU KHIỂN: ESP32-S3 (N16R8) * * TÍNH NĂNG NỔI BẬT: * 1. In nhật ký Serial Monitor đầy đủ, định kỳ mỗi 1s (Dòng RMS, Công suất, Raw ADC, Voltage, Relay). * 2. Tự động Calib điểm 0 (Zero-offset) cho ACS712 khi khởi động. * 3. Gửi Telemetry lên HiveMQ Cloud over Topic độc quyền: smartenergy_node9988/telemetry * 4. Bật/Tắt thủ công Quạt (GPIO 12) và Đèn (GPIO 13) độc lập 100%. * 5. Sửa dứt điểm lỗi biên dịch buffer C++ (Mảng char buffer[2]). * ==================================================================================== */ #include #include #include // ====================================================== // 1. CẤU HÌNH WIFI & CLOUD MQTT BROKER (HIVEMQ PUBLIC) // ====================================================== const char* WIFI_SSID = "HO TRO SINH VIEN"; const char* WIFI_PASSWORD = "12345678@"; const char* MQTT_BROKER = "broker.hivemq.com"; const int MQTT_PORT = 1883; // TOPIC ĐỘC QUYỀN TRÁNH TRÙNG LẶP TRÊN SERVER HIVEMQ CÔNG KHAI const char* CLIENT_ID = "ESP32S3_SmartEnergy_Node_9988_v13"; const char* TOPIC_TELEMETRY = "smartenergy_node9988/telemetry"; const char* TOPIC_CONTROL = "smartenergy_node9988/control"; const char* TOPIC_STATUS = "smartenergy_node9988/status"; // ====================================================== // 2. CHÂN KẾT NỐI PHẦN CỨNG (ESP32-S3 N16R8) // ====================================================== const int PIN_ACS712 = 1; // GPIO 1 (ADC1) đọc cảm biến ACS712 (Cấp nguồn 5V) const int PIN_RELAY_FAN = 12; // GPIO 12 Rơ-le Quạt 12V (IN1 - Active LOW) const int PIN_RELAY_LED = 13; // GPIO 13 Rơ-le Đèn LED Module (IN2 - Active LOW) // ====================================================== // 3. BIẾN HỆ THỐNG & CALIB CẢM BIẾN // ====================================================== float peak_limit = 3.0; // Ngưỡng công suất đỉnh mặc định (W) float hysteresis = 0.5; // Ngưỡng trễ khôi phục (W) bool auto_mode = false; // Mặc định THỦ CÔNG 100% để làm chủ nút bấm trên Web bool fan_state = false; // Trạng thái Quạt (false: TẮT, true: BẬT) bool led_state = false; // Trạng thái Đèn LED (false: TẮT, true: BẬT) bool was_shed_by_auto = false; // Cờ đánh dấu: CHỈ BẬT LẠI NẾU BỊ TẮT DO SA THẢI TẢI float zero_voltage = 2.5; // Điện áp điểm 0 (Vref) của ACS712 int raw_adc_zero = 3100; // ADC điểm 0 mặc định float current_rms = 0.0; float power_watts = 0.0; int raw_adc_val = 0; float sensor_voltage = 0.0; const float VOLTAGE_DC = 12.0; // Điện áp hệ thống DC 12V const float ACS712_SENSITIVITY = 0.185; // Độ nhạy ACS712-05B (185mV/A = 0.185V/A) WiFiClient espClient; PubSubClient mqttClient(espClient); unsigned long last_telemetry_time = 0; unsigned long msg_count = 0; // ====================================================== // HÀM TỰ ĐỘNG CALIB ĐIỂM 0 (ZERO OFFSET) KHI KHỞI ĐỘNG // ====================================================== void calibrateACS712Zero() { Serial.println("\n------------------------------------------------"); Serial.println("⏳ Đang Calib điểm 0 cho ACS712..."); Serial.println("⚠️ VUI LÒNG GIỮ QUẠT & TẢI TẮT TRONG 2 GIÂY!"); long sum_raw = 0; int num_samples = 500; for (int i = 0; i < num_samples; i++) { sum_raw += analogRead(PIN_ACS712); delayMicroseconds(200); } raw_adc_zero = sum_raw / num_samples; zero_voltage = (raw_adc_zero / 4095.0) * 3.3; Serial.print("✅ Calib hoàn tất! ADC Zero Baseline = "); Serial.print(raw_adc_zero); Serial.print(" | Điện áp Vref Zero = "); Serial.print(zero_voltage, 3); Serial.println(" V"); Serial.println("------------------------------------------------\n"); } // ====================================================== // HÀM ĐỌC DÒNG RMS VÀ CÔNG SUẤT TỪ ACS712 (ADC 12-bit) // ====================================================== void readACS712() { long sum_sq_diff = 0; long sum_raw = 0; int samples = 400; for (int i = 0; i < samples; i++) { int raw = analogRead(PIN_ACS712); sum_raw += raw; int diff = raw - raw_adc_zero; sum_sq_diff += (long)diff * diff; delayMicroseconds(100); } raw_adc_val = sum_raw / samples; sensor_voltage = (raw_adc_val / 4095.0) * 3.3; float rms_adc_diff = sqrt((float)sum_sq_diff / samples); float rms_voltage_diff = (rms_adc_diff / 4095.0) * 3.3; // Tính dòng điện RMS (Amperes) current_rms = rms_voltage_diff / ACS712_SENSITIVITY; // Lọc nhiễu nền nếu dòng điện quá nhỏ (< 0.04A ~ 0.48W) if (current_rms < 0.04) { current_rms = 0.0; } power_watts = VOLTAGE_DC * current_rms; } // ====================================================== // HÀM BẮN DỮ LIỆU TELEMETRY VÀ IN LOG SERIAL MONITOR // ====================================================== void publishTelemetry() { msg_count++; StaticJsonDocument<384> doc; doc["current"] = current_rms; doc["power"] = power_watts; doc["fan"] = fan_state; doc["led"] = led_state; doc["auto_mode"] = auto_mode; doc["peak_limit"] = peak_limit; doc["raw_adc"] = raw_adc_val; doc["voltage"] = sensor_voltage; doc["msg_id"] = msg_count; // SỬA LỖI COMPILATION: Khai báo mảng char buffer 384 bytes char buffer[2]; serializeJson(doc, buffer); mqttClient.publish(TOPIC_TELEMETRY, buffer); // --- IN GIÁ TRỊ CHI TIẾT RA SERIAL MONITOR TRÊN ARDUINO IDE --- Serial.print("[#"); Serial.print(msg_count); Serial.print("] ADC Pin1: "); Serial.print(raw_adc_val); Serial.print(" | Vol: "); Serial.print(sensor_voltage, 2); Serial.print("V | Dòng RMS: "); Serial.print(current_rms, 2); Serial.print(" A | Công suất: "); Serial.print(power_watts, 2); Serial.print(" W | Quạt: "); Serial.print(fan_state ? "ON " : "OFF"); Serial.print(" | Đèn: "); Serial.print(led_state ? "ON " : "OFF"); Serial.print(" | Mode: "); Serial.println(auto_mode ? "AUTO" : "MANUAL"); } // ====================================================== // HÀM XỬ LÝ LỆNH ĐIỀU KHIỂN TỪ WEB VERCEL // ====================================================== void mqttCallback(char* topic, byte* payload, unsigned int length) { StaticJsonDocument<256> doc; DeserializationError error = deserializeJson(doc, payload, length); if (error) return; Serial.println("\n📩 [MQTT IN] Nhận lệnh từ Web Vercel:"); if (doc.containsKey("auto_mode")) { auto_mode = doc["auto_mode"]; Serial.print(" -> Chế độ Auto: "); Serial.println(auto_mode ? "BẬT" : "TẮT"); } if (doc.containsKey("peak_limit")) { peak_limit = doc["peak_limit"]; Serial.print(" -> Cài Ngưỡng Đỉnh: "); Serial.print(peak_limit); Serial.println(" W"); } // Lệnh bật/tắt Quạt thủ công if (doc.containsKey("fan")) { fan_state = doc["fan"]; digitalWrite(PIN_RELAY_FAN, fan_state ? LOW : HIGH); // Active LOW Serial.print(" -> Đã chuyển Quạt 12V: "); Serial.println(fan_state ? "BẬT (ON)" : "TẮT (OFF)"); } // Lệnh bật/tắt Đèn LED thủ công if (doc.containsKey("led")) { led_state = doc["led"]; was_shed_by_auto = false; // Xóa cờ sa thải tải khi bấm thủ công digitalWrite(PIN_RELAY_LED, led_state ? LOW : HIGH); // Active LOW Serial.print(" -> Đã chuyển Đèn LED: "); Serial.println(led_state ? "BẬT (ON)" : "TẮT (OFF)"); } publishTelemetry(); } // ====================================================== // KẾT NỐI WIFI & MQTT BROKER TỰ ĐỘNG KHÔI PHỤC (LWT) // ====================================================== void reconnectMQTT() { while (!mqttClient.connected()) { Serial.print("Đang kết nối HiveMQ Broker... "); if (mqttClient.connect(CLIENT_ID, TOPIC_STATUS, 1, true, "OFFLINE")) { Serial.println("THÀNH CÔNG!"); mqttClient.publish(TOPIC_STATUS, "ONLINE", true); mqttClient.subscribe(TOPIC_CONTROL); } else { Serial.print("Thất bại, rc="); Serial.print(mqttClient.state()); Serial.println(". Thử lại sau 2 giây..."); delay(2000); } } } void setup() { Serial.begin(115200); delay(1000); Serial.println("\n===================================================="); Serial.println(" ESP32-S3 SMART ENERGY & LOAD MANAGEMENT SYSTEM "); Serial.println("===================================================="); // Khóa cứng Rơ-le ở mức TẮT (HIGH) tránh giật khi cắm USB digitalWrite(PIN_RELAY_FAN, HIGH); digitalWrite(PIN_RELAY_LED, HIGH); pinMode(PIN_RELAY_FAN, OUTPUT); pinMode(PIN_RELAY_LED, OUTPUT); digitalWrite(PIN_RELAY_FAN, HIGH); digitalWrite(PIN_RELAY_LED, HIGH); analogReadResolution(12); analogSetAttenuation(ADC_11db); // Calib điểm 0 cảm biến ACS712 calibrateACS712Zero(); // Kết nối Wi-Fi Serial.print("Đang kết nối Wi-Fi: "); Serial.println(WIFI_SSID); WiFi.mode(WIFI_STA); WiFi.begin(WIFI_SSID, WIFI_PASSWORD); while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); } Serial.println("\n✅ Wi-Fi đã kết nối thành công!"); Serial.print("📌 Địa chỉ IP ESP32-S3: "); Serial.println(WiFi.localIP()); mqttClient.setServer(MQTT_BROKER, MQTT_PORT); mqttClient.setCallback(mqttCallback); reconnectMQTT(); } void loop() { if (!mqttClient.connected()) reconnectMQTT(); mqttClient.loop(); // Đọc cảm biến và gửi telemetry định kỳ đúng mỗi 1 giây if (millis() - last_telemetry_time >= 1000) { last_telemetry_time = millis(); readACS712(); // THUẬT TOÁN SA THẢI TẢI TỰ ĐỘNG (CHỈ CHẠY KHI BẬT CHẾ ĐỘ AUTO) if (auto_mode) { if (power_watts > peak_limit && led_state) { led_state = false; was_shed_by_auto = true; digitalWrite(PIN_RELAY_LED, HIGH); Serial.println("⚠️ [SA THẢI TẢI] Vượt ngưỡng đỉnh! Tự động TẮT Đèn LED."); } else if (power_watts < (peak_limit - hysteresis) && !led_state && was_shed_by_auto) { led_state = true; was_shed_by_auto = false; digitalWrite(PIN_RELAY_LED, LOW); Serial.println("✅ [SA THẢI TẢI] Công suất an toàn! Tự động BẬT LẠI Đèn LED."); } } publishTelemetry(); } } 2.
